\subsubsection*{Requirements Specifications}
\begin{requirementbox}{GT-FR-1.1}{Functional Requirement}{VERIFIED}
The system shall validate the 16-digit card number using the Luhn algorithm.
\end{requirementbox}

\begin{requirementbox}{FR-CUSTOM-1}{Credit Card Processing}{VERIFIED}
The system shall verify CVV with the card issuing network.
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=4pt]
    \item \textbf{Rationale:} Reduces card-not-present fraud.
    \item \textbf{Fit Criterion:} 99.9\% of invalid CVVs rejected in under 500ms.
    \item \textbf{Verification:} Test
    \item \textbf{Source:} PCI DSS 3.2
\end{itemize}
\end{requirementbox}
