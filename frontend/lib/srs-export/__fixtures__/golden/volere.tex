\section{Functional Requirements}
\subsection{Shift Assignment}
Assign volunteers to open shifts.

\subsubsection*{Requirements Specifications}
\begin{requirementbox}{GT-FR-1.1}{Shift Assignment}{VERIFIED}
The system shall notify a volunteer within 5 minutes of assignment.
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=4pt]
    \item \textbf{Rationale:} Timely confirmation reduces no-shows.
    \item \textbf{Fit Criterion:} 95\% of notifications sent within 5 minutes.
\end{itemize}
\end{requirementbox}

\section{Non-functional Requirements}
\subsection{Look and Feel Requirements}
\begin{requirementbox}{GT-LOO-1}{Look and Feel Requirements}{VERIFIED}
The UI shall use the org's existing brand palette.
\end{requirementbox}

\subsection{Usability and Humanity Requirements}
\begin{requirementbox}{GT-USA-1}{Usability and Humanity Requirements}{VERIFIED}
A first-time coordinator shall complete shift setup within 10 minutes unaided.
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=4pt]
    \item \textbf{Fit Criterion:} 90\% success rate in usability testing.
\end{itemize}
\end{requirementbox}

\subsection{Operational and Environmental Requirements}
